Arahkan kamera ke Barcode atau QR Code

const app = initializeApp(firebaseConfig); const database = getDatabase(app); let isProcessing = false; // Element HTML const video = document.createElement("video"); const canvasElement = document.getElementById("canvas"); const canvas = canvasElement.getContext("2d"); const loadingMessage = document.getElementById("loading-message"); const resultDiv = document.getElementById("result"); function playBeep() { try { const context = new (window.AudioContext || window.webkitAudioContext)(); const oscillator = context.createOscillator(); oscillator.type = 'sine'; oscillator.frequency.setValueAtTime(880, context.currentTime); oscillator.connect(context.destination); oscillator.start(); oscillator.stop(context.currentTime + 0.15); } catch(e) {} } // Minta akses kamera belakang secara paksa (facingMode: environment) navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment" } }).then(function(stream) { video.srcObject = stream; video.setAttribute("playsinline", true); // penting untuk pengguna iOS/iPhone video.play(); requestAnimationFrame(tick); }).catch(function(err) { loadingMessage.innerText = "❌ Gagal memuat kamera. Pastikan izin kamera di browser sudah diaktifkan."; }); // Fungsi Looping untuk membaca frame gambar dari kamera function tick() { if (video.readyState === video.HAVE_ENOUGH_DATA) { loadingMessage.hidden = true; canvasElement.hidden = false; canvasElement.height = video.videoHeight; canvasElement.width = video.videoWidth; canvas.drawImage(video, 0, 0, canvasElement.width, canvasElement.height); // Ambil data gambar dari canvas untuk discan pola QR-nya const imageData = canvas.getImageData(0, 0, canvasElement.width, canvasElement.height); const code = jsQR(imageData.data, imageData.width, imageData.height, { inversionAttempts: "dontInvert", }); // JIKA QR CODE TERDETEKSI OLEH KAMERA if (code && code.data && !isProcessing) { isProcessing = true; // Kunci proses const hasilScan = code.data.trim(); resultDiv.innerHTML = "Memvalidasi kode..."; // Hubungkan ke Firebase jalankan pencocokan data get(ref(database, 'sistem/riwayat_kode')).then((snapshot) => { let matches = false; if (snapshot.exists()) { const semuaRiwayat = snapshot.val(); for (let key in semuaRiwayat) { if (semuaRiwayat[key].angka === hasilScan) { matches = true; break; } } } if (matches) { playBeep(); resultDiv.style.background = "#d4edda"; resultDiv.style.color = "#155724"; resultDiv.innerHTML = `✅ VALID: ${hasilScan}`; setTimeout(() => { alert(`🎉 BERHASIL!\nQR Code terdaftar & sesuai.`); isProcessing = false; }, 200); } else { resultDiv.style.background = "#f8d7da"; resultDiv.style.color = "#721c24"; resultDiv.innerHTML = `❌ GAGAL: ${hasilScan}`; setTimeout(() => { alert(`🚨 GAGAL!\nQR Code tidak ditemukan.`); isProcessing = false; }, 200); } }).catch(() => { isProcessing = false; }); } } requestAnimationFrame(tick); }
